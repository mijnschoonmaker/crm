import { useMemo, useRef, useState } from "react"
import { addPropertyControls, ControlType } from "framer"

type Dot = { x: number; y: number; t: number; id: number }

const SOORTEN = [
    { k: "BG", naam: "Begane grond", kleur: "#e5484d" },
    { k: "1", naam: "1e verdieping", kleur: "#f76b15" },
    { k: "2", naam: "2e verdieping", kleur: "#8e4ec6" },
    { k: "T", naam: "Tuimelraam", kleur: "#0090ff" },
]

const START: [number, number, number][] = [
    [12.7, 67, 0], [38.3, 70, 0], [45.8, 70, 0], [66, 67, 0], [69.3, 72, 0],
    [72.6, 42, 1], [76.2, 42, 1], [79.6, 42, 1],
    [16, 48, 3], [27.3, 47, 3], [39, 44, 3],
]

interface Props {
    foto?: { src?: string; alt?: string } | string
    basis: number
    perRaam: number
    minimum: number
    kleur: string
    lijn: string
    radius: number
    font: any
    style?: React.CSSProperties
}

const euro = (n: number) => "€ " + n.toFixed(2).replace(".", ",")

/**
 * Ramen aantikken op een foto en direct een voorbeeldprijs zien.
 * @framerSupportedLayoutWidth any-prefer-fixed
 * @framerSupportedLayoutHeight auto
 * @framerIntrinsicWidth 560
 */
export default function RamenTikken(props: Props) {
    const { basis = 12, perRaam = 2.6, minimum = 29, kleur = "#121212", lijn = "#EDEDED", radius = 0, font, style } = props
    const src = typeof props.foto === "string" ? props.foto : props.foto?.src || "https://app.mansard.nl/site/huis.jpg"
    const alt = (typeof props.foto === "object" && props.foto?.alt) || "Woning om ramen op aan te tikken"
    const idRef = useRef(100)
    const [dots, setDots] = useState<Dot[]>(() => START.map(([x, y, t], i) => ({ x, y, t, id: i })))
    const [cur, setCur] = useState(0)
    const [fr, setFr] = useState(8)
    const tel = useMemo(() => SOORTEN.map((_, i) => dots.filter((d) => d.t === i).length), [dots])
    const n = dots.length
    const factor = ({ 4: 1, 8: 1.12, 12: 1.25 } as Record<number, number>)[fr]
    const prijs = n ? Math.max(minimum, (basis + n * perRaam) * factor) : 0
    const ff = font?.fontFamily ? font : { fontFamily: "Geist, Inter, system-ui, sans-serif" }
    const r = Math.max(0, radius)

    const tik = (e: React.MouseEvent<HTMLDivElement>) => {
        const b = e.currentTarget.getBoundingClientRect()
        const x = ((e.clientX - b.left) / b.width) * 100
        const y = ((e.clientY - b.top) / b.height) * 100
        setDots((l) => [...l, { x, y, t: cur, id: idRef.current++ }])
    }

    const kop: React.CSSProperties = { fontSize: 11, letterSpacing: "0.18em", textTransform: "uppercase", color: "#8f8f8f" }

    return (
        <div style={{ ...style, ...ff, width: "100%", background: "#fff", border: `1px solid ${lijn}`, borderRadius: r, padding: 14, boxSizing: "border-box", color: kleur }}>
            <div
                onClick={tik}
                role="img"
                aria-label={alt + ". Tik om een raam toe te voegen."}
                style={{ position: "relative", borderRadius: Math.max(0, r - 8), overflow: "hidden", cursor: "crosshair", userSelect: "none", touchAction: "manipulation", lineHeight: 0 }}
            >
                <img src={src} alt="" draggable={false} style={{ width: "100%", display: "block", pointerEvents: "none" }} />
                {dots.map((d) => (
                    <button
                        key={d.id}
                        type="button"
                        aria-label={"Verwijder stip " + SOORTEN[d.t].naam}
                        onClick={(e) => {
                            e.stopPropagation()
                            setDots((l) => l.filter((x) => x.id !== d.id))
                        }}
                        style={{
                            position: "absolute", left: d.x + "%", top: d.y + "%", width: 24, height: 24, margin: "-12px 0 0 -12px",
                            borderRadius: "50%", border: "2px solid #fff", background: SOORTEN[d.t].kleur, color: "#fff",
                            font: "700 9px/1 " + (ff.fontFamily || "sans-serif"), display: "grid", placeItems: "center",
                            boxShadow: "0 3px 10px rgba(0,0,0,.35)", cursor: "pointer", padding: 0,
                        }}
                    >
                        {SOORTEN[d.t].k}
                    </button>
                ))}
            </div>

            <div role="radiogroup" aria-label="Soort raam" style={{ display: "flex", flexWrap: "wrap", gap: 8, padding: "14px 0 4px" }}>
                {SOORTEN.map((s, i) => {
                    const on = i === cur
                    return (
                        <button
                            key={s.k}
                            type="button"
                            role="radio"
                            aria-checked={on}
                            onClick={() => setCur(i)}
                            style={{
                                display: "flex", alignItems: "center", gap: 8, height: 38, padding: "0 14px 0 6px", borderRadius: 99,
                                border: `1px solid ${on ? kleur : lijn}`, background: on ? kleur : "#fff", color: on ? "#fff" : kleur,
                                font: "500 13px " + ff.fontFamily, cursor: "pointer",
                            }}
                        >
                            <i style={{ width: 24, height: 24, borderRadius: "50%", background: s.kleur, color: "#fff", display: "grid", placeItems: "center", font: "700 9px " + ff.fontFamily, fontStyle: "normal" }}>{s.k}</i>
                            {s.naam}
                            <b style={{ opacity: 0.6, fontWeight: 600 }}>{tel[i]}</b>
                        </button>
                    )
                })}
            </div>

            <div style={{ display: "flex", alignItems: "baseline", justifyContent: "space-between", gap: 16, padding: "18px 6px 6px", borderTop: `1px solid ${lijn}`, marginTop: 12 }}>
                <div style={{ fontSize: 38, fontWeight: 300, lineHeight: 1 }}>
                    {n}
                    <small style={{ ...kop, marginLeft: 8 }}>{n === 1 ? "raam" : "ramen"}</small>
                </div>
                <button type="button" onClick={() => setDots([])} style={{ ...kop, background: "none", border: 0, cursor: "pointer", padding: 6, fontFamily: ff.fontFamily }}>
                    Opnieuw
                </button>
            </div>

            <div style={{ display: "flex", flexWrap: "wrap", alignItems: "flex-end", justifyContent: "space-between", gap: 16, padding: "14px 6px 4px", borderTop: `1px solid ${lijn}`, marginTop: 12 }}>
                <div role="group" aria-label="Hoe vaak" style={{ display: "flex", background: "#f4f4f4", borderRadius: Math.min(r, 10), padding: 4 }}>
                    {[4, 8, 12].map((f) => (
                        <button
                            key={f}
                            type="button"
                            aria-pressed={fr === f}
                            onClick={() => setFr(f)}
                            style={{
                                border: 0, borderRadius: Math.min(r, 8), padding: "9px 14px", cursor: "pointer", font: "600 11px " + ff.fontFamily,
                                letterSpacing: "0.16em", textTransform: "uppercase", background: fr === f ? "#fff" : "transparent",
                                color: fr === f ? kleur : "#8f8f8f", boxShadow: fr === f ? "0 1px 3px rgba(0,0,0,.12)" : "none",
                            }}
                        >
                            {f} weken
                        </button>
                    ))}
                </div>
                <div style={{ textAlign: "right" }}>
                    <div style={kop}>Voorbeeldprijs per beurt</div>
                    <div style={{ fontSize: 34, fontWeight: 400, lineHeight: 1.15, fontVariantNumeric: "tabular-nums" }}>{euro(prijs)}</div>
                    <div style={{ fontSize: 12, color: "#8f8f8f" }}>incl. kozijnen en btw</div>
                </div>
            </div>
        </div>
    )
}

addPropertyControls(RamenTikken, {
    foto: { type: ControlType.ResponsiveImage, title: "Foto" },
    basis: { type: ControlType.Number, title: "Starttarief", defaultValue: 12, min: 0, step: 0.5, unit: "€" },
    perRaam: { type: ControlType.Number, title: "Per raam", defaultValue: 2.6, min: 0, step: 0.1, unit: "€" },
    minimum: { type: ControlType.Number, title: "Minimum", defaultValue: 29, min: 0, step: 1, unit: "€" },
    kleur: { type: ControlType.Color, title: "Kleur", defaultValue: "#121212" },
    lijn: { type: ControlType.Color, title: "Lijnen", defaultValue: "#EDEDED" },
    radius: { type: ControlType.Number, title: "Hoeken", defaultValue: 0, min: 0, max: 32 },
    font: { type: ControlType.Font, title: "Lettertype", controls: "basic" },
})
