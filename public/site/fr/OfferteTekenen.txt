import { useEffect, useRef, useState } from "react"
import { addPropertyControls, ControlType } from "framer"

type Dienst = { naam: string; uitleg: string; prijs: string }

interface Props {
    klant: string
    adres: string
    nummer: string
    geldig: string
    diensten: Dienst[]
    logo?: { src?: string; alt?: string } | string
    bedrijf: string
    kleur: string
    lijn: string
    achtergrond: string
    radius: number
    font: any
    style?: React.CSSProperties
}

const STANDAARD: Dienst[] = [
    { naam: "Glasbewassing buiten", uitleg: "14 ramen · elke 8 weken", prijs: "€ 42,50" },
    { naam: "Kozijnreiniging", uitleg: "Rondom · 2× per jaar", prijs: "€ 65,00" },
    { naam: "Dakramen", uitleg: "3 tuimelramen · elke 8 weken", prijs: "€ 9,00" },
]

/**
 * Voorbeeldofferte waarop bezoekers met vinger of muis hun handtekening zetten.
 * @framerSupportedLayoutWidth any-prefer-fixed
 * @framerSupportedLayoutHeight auto
 * @framerIntrinsicWidth 520
 */
export default function OfferteTekenen(props: Props) {
    const {
        klant = "Sanne de Vries", adres = "Lindenlaan 14", nummer = "OFF-2026-014", geldig = "19 nov 2026",
        bedrijf = "", kleur = "#121212", lijn = "#EDEDED", achtergrond = "#121212", radius = 0, font, style,
    } = props
    const diensten = props.diensten && props.diensten.length ? props.diensten : STANDAARD
    const logo = typeof props.logo === "string" ? props.logo : props.logo?.src
    const ff = font?.fontFamily ? font : { fontFamily: "Geist, Inter, system-ui, sans-serif" }
    const r = Math.max(0, radius)

    const cv = useRef<HTMLCanvasElement>(null)
    const tekent = useRef(false)
    const vorige = useRef<[number, number] | null>(null)
    const [heeft, setHeeft] = useState(false)
    const [akkoord, setAkkoord] = useState(false)
    const [klaar, setKlaar] = useState(false)

    const maat = () => {
        const c = cv.current
        if (!c) return
        const b = c.getBoundingClientRect()
        const dpr = Math.max(1, window.devicePixelRatio || 1)
        c.width = Math.max(1, b.width * dpr)
        c.height = Math.max(1, b.height * dpr)
        const ctx = c.getContext("2d")
        if (!ctx) return
        ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
        ctx.lineWidth = 2.2
        ctx.lineCap = "round"
        ctx.lineJoin = "round"
        ctx.strokeStyle = kleur
    }
    useEffect(() => {
        if (klaar) return
        maat()
        const opnieuw = () => { if (!heeftRef.current) maat() }
        window.addEventListener("resize", opnieuw)
        return () => window.removeEventListener("resize", opnieuw)
    }, [klaar, kleur])
    const heeftRef = useRef(false)
    heeftRef.current = heeft

    const punt = (e: React.PointerEvent): [number, number] => {
        const b = cv.current!.getBoundingClientRect()
        return [e.clientX - b.left, e.clientY - b.top]
    }
    const neer = (e: React.PointerEvent<HTMLCanvasElement>) => {
        tekent.current = true
        vorige.current = punt(e)
        e.currentTarget.setPointerCapture(e.pointerId)
    }
    const beweeg = (e: React.PointerEvent<HTMLCanvasElement>) => {
        if (!tekent.current || !vorige.current) return
        const ctx = cv.current?.getContext("2d")
        if (!ctx) return
        const p = punt(e), l = vorige.current
        ctx.beginPath()
        ctx.moveTo(l[0], l[1])
        ctx.quadraticCurveTo(l[0], l[1], (l[0] + p[0]) / 2, (l[1] + p[1]) / 2)
        ctx.lineTo(p[0], p[1])
        ctx.stroke()
        vorige.current = p
        if (!heeft) setHeeft(true)
    }
    const op = () => { tekent.current = false }
    const wis = () => {
        const c = cv.current
        c?.getContext("2d")?.clearRect(0, 0, c.width, c.height)
        setHeeft(false)
    }

    const label: React.CSSProperties = { fontSize: 10, letterSpacing: "0.2em", textTransform: "uppercase", color: "#8f8f8f" }
    const kaart: React.CSSProperties = { background: "#fff", color: kleur }
    const door = (
        <div style={{ textAlign: "center", marginTop: 18 }}>
            <div style={{ ...label, fontSize: 9 }}>Deze service wordt mogelijk gemaakt door</div>
            <div style={{ fontSize: 13, letterSpacing: "0.32em", marginTop: 6, color: "#4b4b4b" }}>MANSARD</div>
        </div>
    )

    return (
        <div style={{ ...style, ...ff, width: "100%", filter: "drop-shadow(0 30px 40px rgba(0,0,0,.18))" }}>
            <div style={{ ...kaart, borderRadius: `${r}px ${r}px 0 0`, overflow: "hidden" }}>
                <div style={{ background: achtergrond, padding: 26, display: "grid", placeItems: "center", position: "relative", minHeight: 46 }}>
                    {logo ? (
                        <img src={logo} alt={bedrijf || "Logo"} style={{ height: 46, width: "auto", display: "block" }} />
                    ) : (
                        <span style={{ color: "#fff", fontSize: 13, letterSpacing: "0.32em", textTransform: "uppercase" }}>{bedrijf || "Uw bedrijf"}</span>
                    )}
                    <span style={{ position: "absolute", right: 16, top: 16, color: "#fff", border: "1px solid rgba(255,255,255,.35)", borderRadius: 99, padding: "5px 11px", fontSize: 9.5, letterSpacing: "0.2em", textTransform: "uppercase" }}>
                        Probeer het
                    </span>
                </div>
                <div style={{ padding: "24px 26px 26px" }}>
                    <div style={{ fontSize: 16, letterSpacing: "0.12em", textTransform: "uppercase", fontWeight: 600 }}>{klant}</div>
                    <div style={{ color: "#8f8f8f", fontSize: 14 }}>{adres}</div>
                    <div style={{ display: "grid", gridTemplateColumns: "repeat(3,minmax(0,1fr))", gap: 8, marginTop: 20 }}>
                        {[["Offerte", nummer], ["Geldig tot", geldig], ["Diensten", String(diensten.length)]].map(([l, w]) => (
                            <div key={l} style={{ background: "#f6f6f6", borderRadius: Math.min(r, 12), padding: "10px 12px", minWidth: 0 }}>
                                <div style={{ ...label, fontSize: 9 }}>{l}</div>
                                <b style={{ display: "block", fontSize: 13, marginTop: 3, whiteSpace: "nowrap", overflow: "hidden", textOverflow: "ellipsis" }}>{w}</b>
                            </div>
                        ))}
                    </div>
                </div>
            </div>

            <div style={{ position: "relative", height: 0, zIndex: 1 }}>
                <div style={{ position: "absolute", left: 26, right: 26, top: -1, borderTop: "2px dashed #d9d9d9" }} />
            </div>

            <div style={{ ...kaart, borderRadius: `0 0 ${r}px ${r}px`, padding: "20px 26px 26px" }}>
                {klaar ? (
                    <div style={{ textAlign: "center", padding: "18px 6px 6px" }}>
                        <div style={{ width: 62, height: 62, borderRadius: "50%", background: "#e8f5ec", color: "#1f7a45", display: "grid", placeItems: "center", margin: "0 auto 14px" }}>
                            <svg width="28" height="28" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.2" strokeLinecap="round" strokeLinejoin="round"><path d="m5 12.5 4.5 4.5L19 7.5" /></svg>
                        </div>
                        <div style={{ fontSize: 16, letterSpacing: "0.14em", textTransform: "uppercase", fontWeight: 500 }}>Uw akkoord is ontvangen</div>
                        <p style={{ color: "#8f8f8f", fontSize: 14, margin: "10px 0 0", lineHeight: 1.5 }}>
                            U ontvangt het definitieve bestellingsoverzicht per e-mail. Binnen 2 tot 8 weken volgt een planningsmail.
                        </p>
                        <button type="button" onClick={() => { setKlaar(false); setHeeft(false); setAkkoord(false) }} style={{ ...label, background: "none", border: 0, cursor: "pointer", marginTop: 14, fontFamily: ff.fontFamily }}>
                            Nog een keer proberen
                        </button>
                        {door}
                    </div>
                ) : (
                    <>
                        {diensten.map((d, i) => (
                            <div key={i} style={{ display: "flex", justifyContent: "space-between", alignItems: "center", gap: 12, padding: "12px 0", borderBottom: i < diensten.length - 1 ? `1px solid ${lijn}` : "0" }}>
                                <div>
                                    <b style={{ display: "block", fontSize: 15, fontWeight: 600 }}>{d.naam}</b>
                                    <small style={{ color: "#8f8f8f", fontSize: 12.5 }}>{d.uitleg}</small>
                                </div>
                                <div style={{ textAlign: "right", fontWeight: 600, whiteSpace: "nowrap", fontVariantNumeric: "tabular-nums" }}>
                                    {d.prijs}
                                    <small style={{ display: "block", fontWeight: 400, fontSize: 10.5, letterSpacing: "0.06em", color: "#8f8f8f" }}>incl. btw</small>
                                </div>
                            </div>
                        ))}

                        <div style={{ marginTop: 14, border: "1px dashed #cfcfcf", borderRadius: Math.min(r, 14), position: "relative", background: "#f8f8f8" }}>
                            <canvas
                                ref={cv}
                                aria-label="Handtekeningveld: teken met uw vinger of muis"
                                onPointerDown={neer}
                                onPointerMove={beweeg}
                                onPointerUp={op}
                                onPointerCancel={op}
                                onPointerLeave={op}
                                style={{ display: "block", width: "100%", height: 130, touchAction: "none", cursor: "crosshair" }}
                            />
                            {!heeft && (
                                <div style={{ position: "absolute", inset: 0, display: "grid", placeItems: "center", pointerEvents: "none", color: "#a3a3a3", fontSize: 11, letterSpacing: "0.24em", textTransform: "uppercase", textAlign: "center" }}>
                                    <div>
                                        <svg width="22" height="22" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="1.6" strokeLinecap="round" strokeLinejoin="round" style={{ display: "block", margin: "0 auto 6px" }}><path d="M4 17c3-1 4-6 7-6s2 5 5 5 3-3 4-4" /><path d="M3 21h18" /></svg>
                                        Teken hier uw handtekening
                                    </div>
                                </div>
                            )}
                            {heeft && (
                                <button type="button" onClick={wis} style={{ position: "absolute", right: 8, top: 8, background: "#fff", border: `1px solid ${lijn}`, borderRadius: 8, fontSize: 11, padding: "4px 9px", cursor: "pointer", color: "#8f8f8f", fontFamily: ff.fontFamily }}>
                                    Wissen
                                </button>
                            )}
                        </div>

                        <label style={{ display: "flex", gap: 10, alignItems: "center", margin: "16px 0 8px", fontSize: 13.5, color: "#4b4b4b", cursor: "pointer" }}>
                            <input type="checkbox" checked={akkoord} onChange={(e) => setAkkoord(e.target.checked)} style={{ width: 18, height: 18, accentColor: kleur, flex: "none" }} />
                            Ik ga akkoord met de offerte en de voorwaarden
                        </label>
                        <p style={{ margin: "0 0 12px", fontSize: 11, color: "#8f8f8f", textAlign: "center" }}>Voorbeeld: deze offerte is niet echt en er wordt niets opgeslagen.</p>
                        <button
                            type="button"
                            disabled={!(heeft && akkoord)}
                            onClick={() => setKlaar(true)}
                            style={{
                                width: "100%", height: 52, border: 0, borderRadius: Math.min(r, 12), background: kleur, color: "#fff",
                                font: "600 13px " + ff.fontFamily, letterSpacing: "0.2em", textTransform: "uppercase",
                                cursor: heeft && akkoord ? "pointer" : "not-allowed", opacity: heeft && akkoord ? 1 : 0.4, transition: "opacity .2s",
                            }}
                        >
                            Akkoord geven
                        </button>
                        {door}
                    </>
                )}
            </div>
        </div>
    )
}

addPropertyControls(OfferteTekenen, {
    klant: { type: ControlType.String, title: "Klant", defaultValue: "Sanne de Vries" },
    adres: { type: ControlType.String, title: "Adres", defaultValue: "Lindenlaan 14" },
    nummer: { type: ControlType.String, title: "Offerte", defaultValue: "OFF-2026-014" },
    geldig: { type: ControlType.String, title: "Geldig tot", defaultValue: "19 nov 2026" },
    diensten: {
        type: ControlType.Array,
        title: "Diensten",
        control: {
            type: ControlType.Object,
            controls: {
                naam: { type: ControlType.String, title: "Dienst", defaultValue: "Glasbewassing" },
                uitleg: { type: ControlType.String, title: "Uitleg", defaultValue: "elke 8 weken" },
                prijs: { type: ControlType.String, title: "Prijs", defaultValue: "€ 40,00" },
            },
        },
        defaultValue: STANDAARD,
    },
    logo: { type: ControlType.ResponsiveImage, title: "Logo" },
    bedrijf: { type: ControlType.String, title: "Bedrijfsnaam", defaultValue: "", placeholder: "Als er geen logo is" },
    kleur: { type: ControlType.Color, title: "Kleur", defaultValue: "#121212" },
    achtergrond: { type: ControlType.Color, title: "Kop", defaultValue: "#121212" },
    lijn: { type: ControlType.Color, title: "Lijnen", defaultValue: "#EDEDED" },
    radius: { type: ControlType.Number, title: "Hoeken", defaultValue: 0, min: 0, max: 32 },
    font: { type: ControlType.Font, title: "Lettertype", controls: "basic" },
})
