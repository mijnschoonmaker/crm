import { useEffect, useRef, useState } from "react"
import { addPropertyControls, ControlType } from "framer"

type Img = { src?: string; alt?: string } | string | undefined
interface Props {
    kaart?: Img
    bus?: Img
    dienst: string
    adres: string
    tijd: string
    style?: React.CSSProperties
}

const ROUTE = "M618.0,700.9 L620.3,700.6 L626.2,696.2 L660.4,659.1 L671.8,647.9 L681.7,647.0 L691.4,647.3 L721.8,612.0 L734.8,596.4 L769.3,557.3 L786.3,560.2 L835.1,504.9 L859.2,478.2 L865.0,467.9 L881.0,469.6 L885.5,462.3 L890.5,453.8 L903.3,431.9 L910.2,420.2 L918.8,405.6 L917.1,394.1 L917.2,385.2 L936.8,358.1 L982.0,299.1"
const FONT = '"Nunito Sans","Avenir Next","Avenir","Segoe UI",system-ui,sans-serif'
const src = (i: Img) => (typeof i === "string" ? i : i?.src) || ""

/**
 * Live kaart: de bus met boxvan rijdt over de route naar de woning, met aftellende adressen.
 * @framerSupportedLayoutWidth any-prefer-fixed
 * @framerSupportedLayoutHeight auto
 * @framerIntrinsicWidth 400
 */
export default function LiveKaart(props: Props) {
    const { dienst = "Glasbewassing", adres = "Lindenlaan 14", tijd = "11:10", style } = props
    const kaart = src(props.kaart) || "https://app.mansard.nl/site/hero-kaart.jpg"
    const bus = src(props.bus) || "https://app.mansard.nl/site/bus.png"
    const route = useRef<SVGPathElement>(null)
    const klaar = useRef<SVGPathElement>(null)
    const busG = useRef<SVGGElement>(null)
    const [f, setF] = useState(0.55)
    const doos = useRef<HTMLDivElement>(null)
    const [smal, setSmal] = useState(false)

    useEffect(() => {
        const R = route.current, D = klaar.current, B = busG.current
        if (!R || !D || !B) return
        const L = R.getTotalLength(), BL = 72, DUR = 24000
        D.style.strokeDasharray = String(L)
        const zet = (v: number) => {
            const d = Math.max(BL, Math.min(L, v * L)), p = R.getPointAtLength(d), a = R.getPointAtLength(Math.max(0, d - BL))
            const m = { x: (p.x + a.x) / 2, y: (p.y + a.y) / 2 }
            let ang = (Math.atan2(p.y - a.y, p.x - a.x) * 180) / Math.PI, flip = 1
            if (Math.abs(ang) > 90) { flip = -1; ang = ang > 0 ? ang - 180 : ang + 180 }
            B.setAttribute("transform", `translate(${m.x},${m.y}) rotate(${ang}) scale(${flip},1)`)
            D.style.strokeDashoffset = String(L - d)
        }
        const stil = typeof window !== "undefined" && window.matchMedia && window.matchMedia("(prefers-reduced-motion: reduce)").matches
        if (stil) { zet(0.55); setF(0.55); return }
        let t0: number | null = null, raf = 0, laatst = -1
        const tik = (t: number) => {
            if (t0 === null) t0 = t
            let p = ((t - t0) % (DUR + 2500)) / DUR
            if (p > 1) p = 1
            const e = p < 0.5 ? 2 * p * p : 1 - Math.pow(-2 * p + 2, 2) / 2
            const v = 0.04 + 0.96 * e
            zet(v)
            const r = Math.round(v * 50) / 50
            if (r !== laatst) { laatst = r; setF(v) }
            raf = requestAnimationFrame(tik)
        }
        raf = requestAnimationFrame(tik)
        return () => cancelAnimationFrame(raf)
    }, [])
    useEffect(() => {
        const meet = () => doos.current && setSmal(doos.current.getBoundingClientRect().width < 380)
        meet()
        window.addEventListener("resize", meet)
        return () => window.removeEventListener("resize", meet)
    }, [])

    const nog = f < 0.45 ? 2 : f < 0.8 ? 1 : 0
    const titel = nog ? `Nog ${nog} ${nog === 1 ? "adres" : "adressen"} voor u` : f < 0.97 ? "Wij zijn onderweg naar u" : "Wij zijn er!"
    const sub = nog === 2 ? "Wij zijn bij u in de buurt" : nog === 1 ? "Bijna bij u" : f < 0.97 ? `Nog ${Math.max(1, Math.round((1 - f) * 6))} minuten` : "Tot zo"
    const glas: React.CSSProperties = {
        position: "absolute", display: "flex", alignItems: "center", gap: 12, background: "rgba(255,255,255,.9)",
        backdropFilter: "blur(14px) saturate(1.3)", WebkitBackdropFilter: "blur(14px) saturate(1.3)", border: "1px solid rgba(255,255,255,.7)",
        borderRadius: 18, boxShadow: "0 24px 40px -24px rgba(0,0,0,.6)", color: "#0d121b",
    }
    const klein: React.CSSProperties = { display: "block", fontSize: 9.5, fontWeight: 700, letterSpacing: "0.22em", textTransform: "uppercase", color: "#6b7180" }
    const stip = <i style={{ width: 7, height: 7, borderRadius: "50%", background: "#3fbf7f", display: "inline-block" }} />

    return (
        <div ref={doos} style={{ ...style, width: "100%", position: "relative", paddingBottom: smal ? 66 : 58, fontFamily: FONT, WebkitFontSmoothing: "antialiased" }}>
            <link rel="stylesheet" href="https://fonts.googleapis.com/css2?family=Nunito+Sans:opsz,wght@6..12,400;6..12,700;6..12,800&display=swap" />
            <div role="img" aria-label="Live kaart: onze bus met boxvan rijdt over de weg naar uw woning" style={{ position: "relative", aspectRatio: smal ? "1 / 0.95" : "10 / 7.6", borderRadius: 26, background: "#3c4636", boxShadow: "0 60px 90px -50px rgba(0,0,0,.85), 0 0 0 1px rgba(255,255,255,.1)" }}>
                <svg viewBox="600 200 560 392" preserveAspectRatio="xMidYMid slice" style={{ position: "absolute", inset: 0, width: "100%", height: "100%", display: "block", borderRadius: 26, clipPath: "inset(0 round 26px)" }} aria-hidden="true">
                    <defs>
                        <radialGradient id="lkVig" cx="50%" cy="50%" r="75%"><stop offset="55%" stopColor="#0d121b" stopOpacity="0" /><stop offset="100%" stopColor="#0d121b" stopOpacity=".35" /></radialGradient>
                        <filter id="lkSch" x="-50%" y="-50%" width="200%" height="200%"><feDropShadow dx="0" dy="4" stdDeviation="4" floodColor="#000" floodOpacity=".45" /></filter>
                    </defs>
                    <image href={kaart} x="0" y="0" width="1600" height="1000" preserveAspectRatio="none" />
                    <rect x="0" y="0" width="1600" height="1000" fill="#1c2231" opacity=".14" />
                    <path d={ROUTE} fill="none" stroke="#fff" strokeOpacity=".35" strokeWidth="9" strokeLinecap="round" strokeLinejoin="round" />
                    <path ref={route} d={ROUTE} fill="none" stroke="#fff" strokeWidth="3.5" strokeDasharray="1 9" strokeLinecap="round" strokeLinejoin="round" />
                    <path ref={klaar} d={ROUTE} fill="none" stroke="#252c3c" strokeOpacity=".9" strokeWidth="5" strokeLinecap="round" strokeLinejoin="round" />
                    <g transform="translate(982,299.1)">
                        <circle r="22" fill="#fff" opacity=".18" />
                        <circle r="12" fill="#252c3c" stroke="#fff" strokeWidth="2.5" filter="url(#lkSch)" />
                        <path d="M-5 1 0-3.6 5 1V5.6H-5z" fill="none" stroke="#fff" strokeWidth="1.6" strokeLinejoin="round" />
                    </g>
                    <g ref={busG} filter="url(#lkSch)"><image href={bus} x="-66" y="-28" width="132" height="34" /></g>
                    <rect x="600" y="200" width="560" height="392" fill="url(#lkVig)" pointerEvents="none" />
                </svg>
                {!smal && (
                    <span style={{ position: "absolute", right: 16, top: 16, display: "flex", alignItems: "center", gap: 7, height: 30, padding: "0 12px", borderRadius: 99, background: "rgba(13,18,27,.55)", backdropFilter: "blur(8px)", WebkitBackdropFilter: "blur(8px)", border: "1px solid rgba(255,255,255,.18)", color: "#fff", fontSize: 10.5, fontWeight: 800, letterSpacing: "0.2em", textTransform: "uppercase" }}>
                        {stip}Live
                    </span>
                )}
                <div style={{ ...glas, left: 14, top: 14, right: smal ? 14 : "auto", padding: "10px 16px 10px 10px" }}>
                    <span style={{ width: 38, height: 38, borderRadius: 11, background: "#252c3c", color: "#fff", display: "grid", placeItems: "center", flex: "none" }}>
                        <svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="1.8" strokeLinecap="round" strokeLinejoin="round"><rect x="3" y="4.5" width="18" height="16.5" rx="2.5" /><path d="M3 10h18M8 2.5v4M16 2.5v4" /></svg>
                    </span>
                    <div style={{ minWidth: 0 }}>
                        <small style={klein}>Vandaag</small>
                        <b style={{ display: "block", fontSize: smal ? 12.5 : 14, fontWeight: 700, whiteSpace: "nowrap", overflow: "hidden", textOverflow: "ellipsis" }}>{dienst} · {adres}</b>
                    </div>
                </div>
            </div>
            <div style={{ ...glas, right: smal ? 10 : 24, left: smal ? 10 : "auto", bottom: 0, width: smal ? "auto" : "min(400px, calc(100% - 48px))", padding: smal ? "12px 14px" : "14px 18px", justifyContent: "space-between", gap: 18, boxShadow: "0 30px 50px -24px rgba(0,0,0,.7)" }}>
                <div style={{ minWidth: 0, flex: 1 }}>
                    <span style={{ display: "flex", alignItems: "center", gap: 6, fontSize: 10, fontWeight: 800, letterSpacing: "0.2em", color: "#2f8a5f", textTransform: "uppercase" }}>{stip}Live</span>
                    <b style={{ display: "block", fontSize: smal ? 15 : 17, fontWeight: 700, marginTop: 4 }}>{titel}</b>
                    <small style={{ display: "block", fontSize: 12.5, color: "#6b7180", marginTop: 1 }}>{sub}</small>
                    <div style={{ height: 4, borderRadius: 4, background: "#e6e8ec", marginTop: 10, overflow: "hidden" }}>
                        <i style={{ display: "block", height: "100%", width: 15 + f * 85 + "%", borderRadius: 4, background: "linear-gradient(90deg,#3a4560,#252c3c)" }} />
                    </div>
                </div>
                <div style={{ textAlign: "right", borderLeft: "1px solid #e3e5e8", paddingLeft: smal ? 14 : 18, flex: "none" }}>
                    <small style={klein}>Verwacht</small>
                    <b style={{ display: "block", fontSize: smal ? 21 : 26, fontWeight: 400, marginTop: 2, fontVariantNumeric: "tabular-nums" }}>{f < 0.97 ? "± " + tijd : "Nu"}</b>
                </div>
            </div>
        </div>
    )
}

addPropertyControls(LiveKaart, {
    dienst: { type: ControlType.String, title: "Dienst", defaultValue: "Glasbewassing" },
    adres: { type: ControlType.String, title: "Adres", defaultValue: "Lindenlaan 14" },
    tijd: { type: ControlType.String, title: "Verwacht", defaultValue: "11:10" },
    kaart: { type: ControlType.ResponsiveImage, title: "Luchtfoto" },
    bus: { type: ControlType.ResponsiveImage, title: "Bus" },
})
